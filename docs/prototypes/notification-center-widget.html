Notification Center Widget Integration Guide

Example of reusing PushNotificationCenter outside the header bell.

1. Markup Structure

Add a container with the required data-* attributes so the widget knows which endpoints, sound, and icons to use.

<section
  id="dashboard-notification-widget"
  data-endpoint="{{ route('file-tracker-dashboard.notifications') }}"
  data-fallback-endpoint="{{ url('api/file-tracker-dashboard/notifications') }}"
  data-sound-url="{{ asset('sound/sound.wav') }}"
  data-icon-url="{{ asset('assets/logo/logo.png') }}"
  data-poll-interval="30000"
>
  <header>
    <h3>Recent Alerts</h3>
    <span id="dashboard-notification-count">0</span>
  </header>
  <div id="dashboard-notification-loading">Loading...</div>
  <ul id="dashboard-notification-list" class="hidden"></ul>
  <p id="dashboard-notification-empty" class="hidden">No alerts.</p>
  <p id="dashboard-notification-error" class="hidden"></p>
</section>
Use the same endpoints as the header bell so permissions remain consistent. The fallback endpoint should be session-authenticated (web route) to avoid API token errors.

2. JavaScript Integration

Instantiate PushNotificationCenter and wire the update/error events to your widget DOM.

document.addEventListener('DOMContentLoaded', () => {
  const widget = document.querySelector('#dashboard-notification-widget');
  if (!widget || !window.PushNotificationCenter) {
    return;
  }

  const list = widget.querySelector('#dashboard-notification-list');
  const loading = widget.querySelector('#dashboard-notification-loading');
  const empty = widget.querySelector('#dashboard-notification-empty');
  const error = widget.querySelector('#dashboard-notification-error');
  const count = widget.querySelector('#dashboard-notification-count');

  const center = window.PushNotificationCenter.create({
    endpoints: [
      widget.dataset.endpoint,
      widget.dataset.fallbackEndpoint,
    ].filter(Boolean),
    pollInterval: Number(widget.dataset.pollInterval || 30000),
    soundUrl: widget.dataset.soundUrl,
    iconUrl: widget.dataset.iconUrl,
  });

  const render = (items) => {
    if (!items.length) {
      list.classList.add('hidden');
      empty.classList.remove('hidden');
      return;
    }

    empty.classList.add('hidden');
    list.classList.remove('hidden');
    list.innerHTML = items
      .map(item => `
        <li>
          <p class="text-sm font-semibold">${item.title || 'Update'}</p>
          <p class="text-xs text-gray-500">${item.body || ''}</p>
        </li>`)
      .join('');
  };

  center
    .on('update', ({ items, unreadCount }) => {
      loading.classList.add('hidden');
      error.classList.add('hidden');
      count.textContent = unreadCount;
      render(items);
    })
    .on('error', (err) => {
      loading.classList.add('hidden');
      error.textContent = err?.message || 'Unable to load alerts.';
      error.classList.remove('hidden');
    })
    .start();
});

4. Best Practices

  • Prime audio/browser permissions with a user gesture before expecting sound or push notifications.
  • Use the same credentials (session-auth routes) across the app to avoid CORS/token issues.
  • Throttle refreshes on visibility changes to reduce unnecessary network traffic.
  • Re-use the utility for other modules by listening to the update event and rendering custom UIs.
