Available Scopes

Scope MethodDescription
mapped()Get records with mapping = 1
unmapped()Get records with mapping = 0
byBatch($batchNo)Get records by batch number
byShelfRack($shelfRack)Get records by shelf/rack location
byYear($year)Get records by specific year
byLanduse($landuse)Get records by land use type
currentYear()Get records from current year

Usage Examples


// Create new grouping record
$grouping = Grouping::create([
    'pseudo_fileno' => 'TEST-001',
    'mls_fileno' => 'MLS-001',
    'mapping' => 1,
    'group' => 'Group A',
    'batch_no' => 'BATCH-001',
    'shelf_rack' => 'A1-B2',
    'year' => 2025,
    'landuse' => 'RESIDENTIAL'
]);

// Query examples
$mapped = Grouping::mapped()->get();
$unmapped = Grouping::unmapped()->get();
$byBatch = Grouping::byBatch('BATCH-001')->get();
$byShelf = Grouping::byShelfRack('A1-B2')->get();
$byYear = Grouping::byYear(2025)->get();
$byLanduse = Grouping::byLanduse('RESIDENTIAL')->get();
$currentYear = Grouping::currentYear()->get();

// Combined queries
$results = Grouping::mapped()
    ->byBatch('BATCH-001')
    ->byYear(2025)
    ->byLanduse('RESIDENTIAL')
    ->where('created_by', 'Musa Ali')
    ->orderBy('date', 'desc')
    ->get();
