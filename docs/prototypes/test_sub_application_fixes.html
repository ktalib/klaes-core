1. Land Use Resolution Test

Issue: "The land use field is required" validation error

Fix: Added resolveLandUseValue() function that checks multiple sources:

function resolveLandUseValue() {
  // For SUA applications, check the hidden field first
  const suaHidden = document.getElementById('sua_land_use_hidden');
  if (suaHidden && suaHidden.value) {
    return suaHidden.value;
  }

  // Check the SUA select field
  const suaSelect = document.getElementById('sua_land_use');
  if (suaSelect && !suaSelect.disabled && suaSelect.value) {
    return suaSelect.value;
  }

  // For regular sub-applications, check mixed land use selector
  const unitSelect = document.getElementById('unit_land_use');
  if (unitSelect && unitSelect.value) {
    return unitSelect.value;
  }

  // Check for any hidden land_use input
  const fallbackInput = document.querySelector('#subApplicationForm input[name="land_use"]');
  if (fallbackInput && fallbackInput.value) {
    return fallbackInput.value;
  }

  // Get land use from URL parameter as fallback
  const urlParams = new URLSearchParams(window.location.search);
  const landUseFromUrl = urlParams.get('landuse');
  if (landUseFromUrl) {
    return landUseFromUrl;
  }

  return '';
}

Integration: Function is called in both form submission and draft autosave
